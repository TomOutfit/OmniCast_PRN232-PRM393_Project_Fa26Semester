% ---------- Caption formatting ----------
% The visible caption label ("Hình"/"Bảng" vs "Figure"/"Table") is set by the
% per-language file omnicast-captions-vn.tex / omnicast-captions-en.tex so
% that a single shared preamble serves all four reports.
\captionsetup[figure]{labelsep=period,font=small}
\captionsetup[table]{labelsep=period,font=small}
\captionsetup[listing]{labelsep=period,font=small}
